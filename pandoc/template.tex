% Define NotoSerif, Noto Sans Symbols, Noto Sans Symbols 2, Noto Sans Math, and NotoEmoji as fallback fonts using luaotfload.
% The engine will check these fonts sequentially when a glyph is missing in the main font.
% We use the 'file:' prefix with the absolute path so LuaLaTeX can locate the specific files in the local directory.

  luaotfload.add_fallback("notofallback", {
    "[/home/michael/dharma/pandoc/fonts/NotoSerif.ttf]:mode=node;",
    "[/home/michael/dharma/pandoc/fonts/NotoSansSymbols.ttf]:mode=node;",
    "[/home/michael/dharma/pandoc/fonts/NotoSansSymbols2.ttf]:mode=node;",
    "[/home/michael/dharma/pandoc/fonts/NotoSansMath.ttf]:mode=node;",
    "[/home/michael/dharma/pandoc/fonts/NotoEmoji.ttf]:mode=node;"
  })
}
